% --- Titres (guide § 2.3.1) ---
% Chapitre  : MAJUSCULES, gras, 13 pt
% Section   : gras, minuscule, 13 pt
% Sous-sect.: italique, gras, minuscule, 12 pt
% Paragraphe: italique, minuscule, 12 pt
\titleformat{\chapter}[hang]
  {\normalfont\bfseries\fontsize{13}{16}\selectfont\filcenter}
  {\MakeUppercase{\chaptername}~\thechapter~:}{0.4em}{\MakeUppercase}

  {\normalfont\bfseries\fontsize{13}{16}\selectfont}
